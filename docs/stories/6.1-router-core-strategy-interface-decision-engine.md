**Epic Goal**: 实现 3 种路由策略 + 自动 failover + A/B 模式
**Epic DoD relevant to this Story**: "用户可选 quality / cost / latency 三种策略" — Story 6.1 lays the foundational interface + decision engine + 3 strategy IMPL stubs that Stories 6.2-6.5 build on.
```proto
// packages/proto/he/routing/v1/routing.proto
syntax = "proto3";
package he.routing.v1;
option go_package "github.com/.../gen/go/he/routing/v1;routingv1";

service RoutingService {
  rpc SelectModel(SelectModelRequest) returns (SelectModelResponse);
}

enum Strategy {                 // Q-D: enum not string (Story 2.6 R-2 cascade)
  STRATEGY_UNSPECIFIED = 0;     // Q-D: engine treats 0 ≡ DEFAULT(1); never errors
  STRATEGY_DEFAULT     = 1;
  STRATEGY_QUALITY     = 2;
  STRATEGY_COST        = 3;
  STRATEGY_LATENCY     = 4;
}

message SelectModelRequest {
  string   user_id          = 1;
  string   requested_model  = 2;
  Strategy strategy         = 3;   // Q-D enum upgrade (§5.2 sketch had string)
  repeated string ab_models = 4;
  string   he_request_id    = 5;   // Q-H: explicit field (Story 3.6 correlation); slog-only echo
}

message SelectModelResponse {
  string   selected_model       = 1;
  string   adapter_endpoint     = 2;   // Q-K: RESERVED-for-future in 6.1 — always empty; gateway resolves via adapterclient.Registry. Field number held stable.
  bool     is_ab_test           = 3;
  repeated string ab_selected_models = 4;  // additive (Story 6.4 needs it; shape defined now to avoid a 6.4 contract break)
  Strategy strategy_used        = 5;   // Q-I: echoes which strategy actually fired (detects unknown-slug→default fallback)
}
```
